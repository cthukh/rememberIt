ALTER SESSION SET "_ORACLE_SCRIPT"=true;
-- USER SQL
CREATE USER "REMEMBER_IT" IDENTIFIED BY "Remember_It2026"  
DEFAULT TABLESPACE "USERS"
TEMPORARY TABLESPACE "TEMP";

-- QUOTAS
ALTER USER "REMEMBER_IT" QUOTA UNLIMITED ON "USERS";

-- ROLES
GRANT "CONNECT" TO "REMEMBER_IT" ;
GRANT "RESOURCE" TO "REMEMBER_IT" ;
ALTER USER "REMEMBER_IT" DEFAULT ROLE "CONNECT","RESOURCE";
